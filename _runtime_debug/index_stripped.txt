async function enterApp(){
  const v = document.getElementById('inviteCode').value.trim().toUpperCase();
  if(v.length < 6) return toast('Davet kodunu kontrol et');

  if(v === 'PAIRA2026'){
    historyStack = ['home'];
    show('home');
    return;
  }

  let rec = null;

  // Web davetlerinde daima Firebase'deki en güncel kaydı kullan.
  // Eski telefon localStorage verisi, aktifleşmiş hesabın üstüne yazmaya çalışmasın.
  try{
    if(!window.PAIRA_FIREBASE_API){
      await new Promise((resolve,reject)=>{
        const t=setTimeout(()=>reject(new Error('Firebase hazır değil')),5000);
        window.addEventListener('paira-firebase-ready',()=>{clearTimeout(t);resolve();},{once:true});
      });
    }
    rec = await window.PAIRA_FIREBASE_API.getInvitation(v);
    if(rec && rec.status !== 'removed'){
      state.adminMembers = state.adminMembers || [];
      const idx = state.adminMembers.findIndex(m => (m.inviteCode || '').toUpperCase() === v);
      if(idx >= 0) state.adminMembers[idx] = rec;
      else state.adminMembers.unshift(rec);
    }
  }catch(err){
    console.warn('PAIRA invite lookup', err);
    // Yalnız yerel dosya testlerinde eski local kayıtla devam etmeye izin ver.
    if(location.protocol === 'file:' || location.protocol === 'content:'){
      rec = (state.adminMembers || []).find(m => (m.inviteCode || '').toUpperCase() === v && m.status !== 'removed');
    }
    if(!rec){
      toast('Davet Firebase’den doğrulanamadı');
      return;
    }
  }

  if(!rec || rec.status === 'removed'){
    toast('Davet kodu geçersiz');
    return;
  }

  rec.partnerPhones = rec.partnerPhones || [rec.contact || '', rec.contact2 || ''];
  rec.partnerDevices = rec.partnerDevices || ['', ''];

  // Daha önce onaylanmış telefon tekrar davet kodu seçmek zorunda kalmasın.
  // Aktif çift hesabında cihaz hangi partnere kayıtlıysa doğrudan o kimlikle aç.
  const returningDeviceId = getDeviceId();
  const returningPartnerIndex = rec.partnerDevices.findIndex(d => d === returningDeviceId);
  const invitationAlreadyActive = rec.status === 'active' || rec.partnerDevices.filter(Boolean).length >= 2;
  if(invitationAlreadyActive){
    if(returningPartnerIndex >= 0){
      pendingInvitePartner = returningPartnerIndex + 1;
      setDevicePartnerIdentity(rec, returningPartnerIndex);
      state.adminMembers = state.adminMembers || [];
      const activeIdx = state.adminMembers.findIndex(m => (m.inviteCode || m._firebaseId || '').toUpperCase() === v);
      if(activeIdx >= 0) state.adminMembers[activeIdx] = rec;
      else state.adminMembers.unshift(rec);
      persist();
      historyStack = ['home'];
      show('home');
      ensureSharedProfileSync();
      toast('Ortak hesap açıldı ✓');
      return;
    }
    toast('Bu çift hesabı iki onaylı telefona bağlı. Onaylı telefonlardan giriş yapın.');
    return;
  }

  if(!pendingInvitePartner){
    chooseInvitePartner(rec, true);
    return;
  }

  const pIndex = pendingInvitePartner - 1;
  const deviceId = returningDeviceId;

  // Partner onayını atomik transaction ile yap. Böylece iki telefon birbirinin
  // onayını eski veriyle ezemez ve aynı anda onay verseler bile 2/2 korunur.
  try{
    if(!window.PAIRA_FIREBASE_API) throw new Error('Firebase hazır değil');
    rec = await window.PAIRA_FIREBASE_API.approveInvitationPartner(
      rec.inviteCode || rec._firebaseId,
      pIndex,
      deviceId
    );
  }catch(err){
    const msg = String(err?.message || err || '');
    console.warn('PAIRA invite approval transaction', err);
    if(msg.includes('DEVICE_OTHER_PARTNER')){
      toast('Bu telefon diğer partner için kayıtlı. Kendi partner linkinizi açın.');
    }else if(msg.includes('PARTNER_ALREADY_OTHER_DEVICE')){
      toast('Bu partner başka bir telefonda zaten onaylandı.');
    }else if(msg.includes('INVITE_REMOVED')){
      toast('Bu davet artık aktif değil.');
    }else{
      toast('Onay Firebase’e kaydedilemedi');
    }
    return;
  }

  rec.partnerPhones = rec.partnerPhones || [rec.contact || '', rec.contact2 || ''];
  rec.partnerDevices = rec.partnerDevices || ['', ''];
  setDevicePartnerIdentity(rec, pIndex);

  // Ekrandaki yerel kaydı da Firebase'deki son durumla eşitle.
  state.adminMembers = state.adminMembers || [];
  const recIdx = state.adminMembers.findIndex(m => (m.inviteCode || m._firebaseId || '').toUpperCase() === v);
  if(recIdx >= 0) state.adminMembers[recIdx] = rec;
  else state.adminMembers.unshift(rec);

  const bothApproved = rec.partnerDevices.filter(Boolean).length >= 2;
  persist();
  renderAdminMembers();

  if(!bothApproved){
    modal(`<h2 style="color:#e7bd72;margin-top:0">Onayınız Tamam ✓</h2>
      <div class="waiting-card">
        <b>${escapeHtml((rec.partnerNames || [])[pIndex] || 'Partner')}</b>
        <p class="muted">Sizin telefonunuz onaylandı.</p>
        <p style="margin-bottom:0">Diğer partner kendi telefonundan onay verince çift hesabı açılacak.</p>
      </div>
      <button class="gold" style="margin-top:12px" onclick="closeModal()">Tamam</button>`);
    return;
  }

  historyStack = ['home'];
  show('home');
  ensureSharedProfileSync();
  toast('İki partner de onay verdi. Ortak hesap açıldı ✓');
}

function memberCardFromProfile(profile){
  const p = profile || {};
  const id = String(p._firebaseId || '').trim();
  if(!id) return null;
  const photos = Array.isArray(p.photos) ? p.photos.filter(Boolean) : [];
  const coverIndex = Math.max(0, Math.min(Number.isInteger(p.coverIndex) ? p.coverIndex : 0, Math.max(photos.length-1,0)));
  return {
    id,
    name:String(p.name || 'PAIRA Üyesi').trim() || 'PAIRA Üyesi',
    city:String(p.city || '').trim(),
    img:photos[coverIndex] || '',
    profile:p
  };
}

function realMembersData(){
  const self = currentCoupleMember();
  const byId = new Map();
  (directoryMembers || []).forEach(m=>{
    if(m && m.id) byId.set(String(m.id), m);
  });
  // Kendi profilimiz Firestore anlık görüntüsünden daha yeniyse yerel ortak profili kullan.
  byId.set(String(self.id), self);
  return [...byId.values()].sort((a,b)=>{
    if(a.isSelf && !b.isSelf) return -1;
    if(b.isSelf && !a.isSelf) return 1;
    return String(a.name || '').localeCompare(String(b.name || ''), 'tr');
  });
}

async function applyMembersDirectorySnapshot(rows, seq){
  try{
    const api = await waitForPairaFirebaseApi();
    const checked = await Promise.all((rows || []).map(async profile=>{
      const id = String(profile?._firebaseId || '').trim();
      if(!id) return null;
      try{
        const inv = await api.getInvitation(id);
        if(!inv) return null;
        const devices = Array.isArray(inv.partnerDevices) ? inv.partnerDevices.filter(Boolean) : [];
        const active = inv.status !== 'removed' && (inv.status === 'active' || devices.length >= 2);
        return active ? memberCardFromProfile(profile) : null;
      }catch(_){
        return null;
      }
    }));
    if(seq !== membersDirectorySeq) return;
    directoryMembers = checked.filter(Boolean);
    membersDirectoryReady = true;
    const active = document.querySelector('.screen.active');
    if(active && active.id === 'members') renderMembersView();
    if(active && active.id === 'member') renderSelectedMemberView();
  }catch(err){
    console.warn('PAIRA member directory filter', err);
  }
}

async function ensureMembersDirectorySync(){
  if(membersDirectoryUnsub) return;
  try{
    const api = await waitForPairaFirebaseApi();
    membersDirectoryUnsub = api.subscribeCoupleProfiles((rows)=>{
      const seq = ++membersDirectorySeq;
      applyMembersDirectorySnapshot(rows, seq);
    },(err)=>{
      membersDirectoryReady = true;
      console.warn('PAIRA member directory sync', err);
      const active = document.querySelector('.screen.active');
      if(active && active.id === 'members') renderMembersView();
    });
  }catch(err){
    membersDirectoryReady = true;
    console.warn('PAIRA member directory start', err);
  }
}

function waitForPairaFirebaseApi(){
  if(window.PAIRA_FIREBASE_API) return Promise.resolve(window.PAIRA_FIREBASE_API);
  return new Promise((resolve,reject)=>{
    const timer=setTimeout(()=>reject(new Error('Firebase hazır değil')),7000);
    window.addEventListener('paira-firebase-ready',()=>{
      clearTimeout(timer);
      resolve(window.PAIRA_FIREBASE_API);
    },{once:true});
  });
}

async function startRealtimeChat(kind, roomId, boxId){
  const key=`${kind}:${roomId}:${boxId}`;
  if(realtimeChatKey===key && realtimeChatUnsub) return;

  realtimeChatKey=key;
  const box=document.getElementById(boxId);
  if(box) box.innerHTML='<div class="muted" style="padding:12px 2px">Sohbet yükleniyor...</div>';

    const api=await waitForPairaFirebaseApi();

      msgs=>renderRealtimeMessages(boxId,msgs),
      err=>{
        console.warn('PAIRA realtime chat',err);
        if(box) box.innerHTML='<div class="muted" style="padding:12px 2px">Chat bağlantısı kurulamadı.</div>';

    console.warn('PAIRA realtime chat start',err);
    if(box) box.innerHTML='<div class="muted" style="padding:12px 2px">Chat bağlantısı kurulamadı.</div>';
  }
}

function notificationTimeText(ts){
  const n = Number(ts || 0);
  if(!n) return '';
  try{
    return new Date(n).toLocaleString('tr-TR',{day:'2-digit',month:'2-digit',hour:'2-digit',minute:'2-digit'});
  }catch(_){ return ''; }
}

function roomTitle(kind, roomId){
  if(kind === 'event') return (events[roomId] && events[roomId].title) || 'Etkinlik Sohbeti';
  if(kind === 'club') return (clubs[roomId] && clubs[roomId].title) || 'Kulüp Sohbeti';
  if(kind === 'general') return 'PAIRA Genel Sohbet';
  return 'Sohbet';
}

function buildNotificationFeed(){
  const myDevice = getDeviceId();
  const systemRows = (systemNotificationRows || []).map(n=>({
    id:'sys_'+String(n.id || n._firebaseId || n.clientAt || Math.random()),
    source:'system',
    type:String(n.type || 'announcement'),
    title:String(n.title || 'PAIRA Bildirimi'),
    body:String(n.body || ''),
    eventId:String(n.eventId || ''),
    clientAt:Number(n.clientAt || 0)
  }));

  const chatRows = [];
  chatNotificationRows.forEach((messages,key)=>{
    const [kind,roomId] = key.split(':');
    (messages || []).forEach(m=>{
      const ts = Number(m.clientAt || 0);
      if(!ts || ts < notificationStartedAt) return;
      if(String(m.deviceId || '') === String(myDevice || '')) return;
      chatRows.push({
        id:'chat_'+kind+'_'+roomId+'_'+String(m.id || ts),
        source:'chat',
        type:'message',
        kind,
        roomId,
        title:`💬 ${roomTitle(kind,roomId)}`,
        body:`${String(m.partnerName || 'Partner')} • ${String(m.coupleName || 'Çift')}: ${String(m.text || '')}`,
        clientAt:ts
      });
    });
  });

  const seen = new Set();
  currentNotificationFeed = [...systemRows,...chatRows]
    .filter(n=>n.clientAt > 0)
    .sort((a,b)=>b.clientAt-a.clientAt)
    .filter(n=>{ if(seen.has(n.id)) return false; seen.add(n.id); return true; })
    .slice(0,120);
  return currentNotificationFeed;
}

function updateNotificationCounters(){
  ensureHomeNotificationBadge();
  const feed = buildNotificationFeed();
  const unread = feed.filter(n=>n.clientAt > notificationLastRead).length;
  const a = document.getElementById('notificationUnreadCount');
  if(a) a.textContent = unread;
  const b = document.getElementById('homeNotificationBadge');
  if(b){
    b.textContent = unread > 99 ? '99+' : String(unread);
    b.classList.toggle('on', unread > 0);
  }
}

function renderNotifications(){
  updateNotificationCounters();
  const wrap = document.getElementById('notificationsList');
  if(!wrap) return;
  const feed = currentNotificationFeed;
  if(!feed.length){
    wrap.innerHTML = '<div class="notification-empty">Henüz yeni bildirim yok.</div>';
    return;
  }
  wrap.innerHTML = feed.map((n,i)=>{
    const unread = n.clientAt > notificationLastRead;
    const icon = n.type === 'event_new' ? '🗓' : n.type === 'event_update' ? '✏️' : n.type === 'announcement' ? '📣' : '';
    const title = icon && !String(n.title).startsWith(icon) ? `${icon} ${n.title}` : n.title;
    return `<button class="listbtn notification-item ${unread?'unread':''}" onclick="openNotificationItem(${i})">
      <b>${escapeHtml(title)}</b><br>
      <span class="muted">${escapeHtml(n.body || '')}</span>
      <span class="notification-time">${escapeHtml(notificationTimeText(n.clientAt))}</span>
      ${unread?'<span class="notification-dot"></span>':''}
    </button>`;
  }).join('');
}

function openNotificationItem(index){
  const n = currentNotificationFeed[index];
  if(!n) return;
  if(n.source === 'chat'){
    if(n.kind === 'event' && events[n.roomId]) return openEventChat(n.roomId);
    if(n.kind === 'club' && clubs[n.roomId]) return openClub(n.roomId);
    return show('chats');
  }
  if((n.type === 'event_new' || n.type === 'event_update')){
    if(n.eventId && events[n.eventId]) return openEvent(n.eventId);
    show('events');
    toast('Etkinlikler ekranı açıldı');
    return;
  }
}

function stopNotificationChatSubscriptions(){
  chatNotificationUnsubs.forEach(fn=>{ try{ fn(); }catch(_){} });
  chatNotificationUnsubs = [];
  chatNotificationRows.clear();
}

async function startNotificationChatSubscriptions(api){
  stopNotificationChatSubscriptions();
  const rooms = [['general','main']];
  Object.keys(events || {}).forEach(id=>rooms.push(['event',id]));
  Object.keys(clubs || {}).forEach(id=>rooms.push(['club',id]));
  rooms.forEach(([kind,roomId])=>{
    try{
      const unsub = api.subscribeChatMessages(kind,roomId,(rows)=>{
        chatNotificationRows.set(`${kind}:${roomId}`, rows || []);
        updateNotificationCounters();
        const active = document.querySelector('.screen.active');
        if(active && active.id === 'notifications') renderNotifications();
      },err=>console.warn('PAIRA notification chat sync',kind,roomId,err));
      chatNotificationUnsubs.push(unsub);
    }catch(err){ console.warn('PAIRA notification room start',kind,roomId,err); }
  });
}

async function ensureNotificationsSync(){
  if(notificationsSyncStarted) return;
  notificationsSyncStarted = true;
  try{
    const api = await waitForPairaFirebaseApi();
    systemNotificationUnsub = api.subscribeSystemNotifications((rows)=>{
      systemNotificationRows = rows || [];
      updateNotificationCounters();
      const active = document.querySelector('.screen.active');
      if(active && active.id === 'notifications') renderNotifications();
    },err=>{
      console.warn('PAIRA system notification sync',err);
      notificationsSyncStarted = false;
    });
    await startNotificationChatSubscriptions(api);
  }catch(err){
    console.warn('PAIRA notification sync start',err);
    notificationsSyncStarted = false;
  }
}

async function refreshNotificationChatSubscriptions(){

  }catch(err){ console.warn('PAIRA notification room refresh',err); }
}

function submitSuggestionForm(type){
  const labels = {event:'Etkinlik Önerisi', friend:'Arkadaş Önerisi', rating:'Değerlendirme', complaint:'Şikâyet'};

  if(type === 'event'){
    data = {title:getField('sf_title'), city:getField('sf_city'), date:getField('sf_date'), people:getField('sf_people'), note:getField('sf_note')};
    if(!data.title || !data.note) return toast('Etkinlik adı ve açıklama gerekli');

    data = {name:getField('sf_name'), city:getField('sf_city'), contact:getField('sf_contact'), note:getField('sf_note')};

    id: Date.now(),
    type: type,
    label: labels[type],
    sender: currentUser,
    createdAt: new Date().toLocaleString('tr-TR'),
    data: data
  });
  persist();
  closeModal();
  updateAdminCounters();
  toast('Yöneticiye gönderildi ✓');
}

function formatSubmission(item){
  const d = item.data || {};
  let body = '';
  if(item.type === 'event'){
    body = `Etkinlik: ${d.title || '-'}\nŞehir: ${d.city || '-'}\nTarih: ${d.date || '-'}\nKişi: ${d.people || '-'}\n\n${d.note || ''}`;
  } else if(item.type === 'friend'){
    body = `Çift: ${d.name || '-'}\nŞehir: ${d.city || '-'}\nİletişim: ${d.contact || '-'}\n\n${d.note || ''}`;
  } else if(item.type === 'rating'){
    body = `Puan: ${d.rating || '-'}\n\n${d.note || ''}`;
  } else {
    body = `Konu: ${d.subject || '-'}\nİlgili: ${d.related || '-'}\n\n${d.note || ''}`;
  }
  return `<div class="admin-card">
    <div class="type">${item.label}</div>
    <div class="meta">${item.sender} • ${item.createdAt}</div>
    <div class="body">${escapeHtml(body)}</div>
  </div>`;
}

function escapeHtml(s){
  return String(s).replace(/[&<>"']/g, ch => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[ch]));
}

function sharedProfileAccountId(){
  const p = getDevicePartnerIdentity();
  // Ortak profil için iki telefonda da kesin aynı olan anahtar davet kodudur.
  // Eski sürümlerde memberId öne alındığı için aynı çift iki farklı profile düşebiliyordu.
  return String((p && (p.inviteCode || p.memberId)) || '').trim().toUpperCase();
}

function sharedProfilePayload(){
  const p = state.profile || {};
  const me = currentChatIdentity();
  return {
    name:String(p.name || me.coupleName || '').slice(0,120),
    city:String(p.city || '').slice(0,80),
    age1:'',
    age2:'',
    children:String(p.children || '').slice(0,40),
    about:String(p.about || '').slice(0,1200),
    interests:Array.isArray(p.interests) ? p.interests.map(x=>String(x).slice(0,60)).slice(0,30) : [],
    photos:Array.isArray(p.photos) ? p.photos.filter(x=>/^https?:\/\//i.test(String(x))).slice(0,5) : [],
    coverIndex:Number.isInteger(p.coverIndex) ? p.coverIndex : 0,
    updatedAt:Date.now(),
    updatedBy:String(me.partnerName || 'Partner').slice(0,80)
  };
}

function applySharedProfile(remote){
  if(!remote || typeof remote !== 'object') return;
  const local = state.profile || {};
  state.profile = {
    ...local,
    name:remote.name ?? local.name,
    city:remote.city ?? local.city,
    children:remote.children ?? local.children,
    about:remote.about ?? local.about,
    interests:Array.isArray(remote.interests) ? [...remote.interests] : (local.interests || []),
    photos:Array.isArray(remote.photos) ? [...remote.photos] : (Array.isArray(local.photos) ? local.photos : []),
    coverIndex:Number.isInteger(remote.coverIndex) ? remote.coverIndex : (typeof local.coverIndex === 'number' ? local.coverIndex : 0)
  };
  try{ persist(); }catch(_){ }

  const active = document.querySelector('.screen.active');
  if(active && active.id === 'profile') renderProfileView();
  if(active && active.id === 'gallery') renderGalleryView();
  if(active && active.id === 'members') renderMembersView();
  if(active && active.id === 'member') renderSelectedMemberView();
}

function stopSharedProfileSync(){
  if(typeof sharedProfileUnsub === 'function'){
    try{ sharedProfileUnsub(); }catch(_){ }
  }
  sharedProfileUnsub = null;
  sharedProfileKey = '';
}

async function ensureSharedProfileSync(){
  const accountId = sharedProfileAccountId();
  if(!accountId) return;
  if(sharedProfileKey === accountId && sharedProfileUnsub) return;

  stopSharedProfileSync();
  sharedProfileKey = accountId;

  try{
    const api = await waitForPairaFirebaseApi();
    const existing = await api.getCoupleProfile(accountId);
    if(existing){
      applySharedProfile(existing);
    }else{
      await api.saveCoupleProfile(accountId, sharedProfilePayload());
    }
    sharedProfileUnsub = api.subscribeCoupleProfile(accountId, applySharedProfile, err=>{
      console.warn('PAIRA shared profile sync', err);
    });
  }catch(err){
    console.warn('PAIRA shared profile start', err);
  }
}

const PROFILE_INTERESTS = [
  "Gurme","Seyahat","Canlı Müzik","Deniz","Sanat","Wellness",
  "Hafta Sonu","Doğa","Spor","Dans","Tiyatro","Şarap Tadımı"
];
let editingInterests = [];
let editingPhotos = [];
let editingCoverIndex = 0;

function defaultProfilePhoto(){
  return '';
}

function loadProfileEdit(){
  const p = state.profile || {};
  document.getElementById('pe_name').value = p.name || '';
  document.getElementById('pe_city').value = p.city || '';
  document.getElementById('pe_children').value = p.children || '';
  document.getElementById('pe_about').value = p.about || '';
  editingInterests = [...(p.interests || [])];
  editingPhotos = [...(p.photos || [])];
  editingCoverIndex = Math.min(p.coverIndex || 0, Math.max(0, editingPhotos.length-1));
  renderInterestEditor();
  renderPhotoEditor();
}

function renderInterestEditor(){
  const wrap = document.getElementById('pe_interests');
  if(!wrap) return;
  wrap.innerHTML = PROFILE_INTERESTS.map(x => `
    <button type="button" class="interest-chip ${editingInterests.includes(x)?'on':''}" onclick="toggleInterest('${x}')">${x}</button>
  `).join('');
}

function toggleInterest(name){
  if(editingInterests.includes(name)){
    editingInterests = editingInterests.filter(x=>x!==name);
  }else{
    editingInterests.push(name);
  }
  renderInterestEditor();
}

function handleProfilePhotos(event){
  const files = Array.from(event.target.files || []);
  if(!files.length) return;
  const remaining = Math.max(0, 5 - editingPhotos.length);
  if(remaining <= 0){
    toast('En fazla 5 fotoğraf eklenebilir');
    event.target.value = '';
    return;
  }
  const chosen = files.slice(0, remaining);
  let completed = 0;

  chosen.forEach(file=>{
    const reader = new FileReader();
    reader.onload = e=>{
      compressProfileImage(e.target.result, compressed=>{
        editingPhotos.push(compressed);
        completed++;
        if(completed === chosen.length){
          if(editingPhotos.length === chosen.length) editingCoverIndex = 0;
          renderPhotoEditor();
          event.target.value = '';
          if(files.length > remaining) toast('5 fotoğraf sınırı uygulandı');
        }
      });
    };
    reader.readAsDataURL(file);
  });
}

function compressProfileImage(dataUrl, done){
  const img = new Image();
  img.onload = ()=>{
    const max = 900;
    let w = img.width, h = img.height;
    if(Math.max(w,h) > max){
      const r = max / Math.max(w,h);
      w = Math.round(w*r); h = Math.round(h*r);
    }
    const canvas = document.createElement('canvas');
    canvas.width = w; canvas.height = h;
    canvas.getContext('2d').drawImage(img,0,0,w,h);
    done(canvas.toDataURL('image/jpeg',0.76));
  };
  img.src = dataUrl;
}

function renderPhotoEditor(){
  const grid = document.getElementById('pe_photo_grid');
  if(!grid) return;
  if(!editingPhotos.length){
    grid.innerHTML = '<div class="muted" style="grid-column:1/-1;padding:12px 0">Henüz fotoğraf eklenmedi.</div>';
    return;
  }
  grid.innerHTML = editingPhotos.map((src,i)=>`
    <div class="photo-card">
      <img src="${src}" alt="Profil fotoğrafı ${i+1}">
      <button type="button" class="cover" onclick="setCoverPhoto(${i})">${i===editingCoverIndex?'★ Ana':'☆ Ana yap'}</button>
      <button type="button" class="remove" onclick="removeProfilePhoto(${i})">×</button>
    </div>
  `).join('');
}

function setCoverPhoto(i){
  editingCoverIndex = i;
  renderPhotoEditor();
  toast('Ana fotoğraf seçildi');
}

function removeProfilePhoto(i){
  editingPhotos.splice(i,1);
  if(editingCoverIndex >= editingPhotos.length) editingCoverIndex = Math.max(0, editingPhotos.length-1);
  if(i < editingCoverIndex) editingCoverIndex--;
  renderPhotoEditor();
}

async function saveProfileEdit(){
  const name = document.getElementById('pe_name').value.trim();
  const city = document.getElementById('pe_city').value.trim();
  if(!name){
    toast('Çift adı gerekli');
    return;
  }
  state.profile = {
    name,
    city,
    age1:'',
    age2:'',
    children:document.getElementById('pe_children').value,
    about:document.getElementById('pe_about').value.trim(),
    interests:[...editingInterests],
    photos:[...editingPhotos],
    coverIndex:editingCoverIndex
  };
  try{
    persist();
  }catch(err){
    toast('Fotoğraflar çok büyük. Birkaçını azaltıp tekrar deneyin.');
    return;
  }

  const accountId = sharedProfileAccountId();
  if(accountId){
    try{
      const api = await waitForPairaFirebaseApi();
      const uploadedPhotos = await api.uploadCoupleProfilePhotos(accountId, state.profile.photos || []);
      state.profile.photos = uploadedPhotos;
      editingPhotos = [...uploadedPhotos];
      if(state.profile.coverIndex >= uploadedPhotos.length){
        state.profile.coverIndex = Math.max(0, uploadedPhotos.length - 1);
        editingCoverIndex = state.profile.coverIndex;
      }
      persist();
      await api.saveCoupleProfile(accountId, sharedProfilePayload());
    }catch(err){
      console.warn('PAIRA shared profile save', err);
      toast('Profil veya fotoğraflar kaydedilemedi');
      return;
    }
  }

  renderProfileView();
  toast('Ortak profil kaydedildi ✓');
  setTimeout(()=>show('profile'),350);
}

let eventsRealtimeUnsub = null;
let eventsSyncStarted = false;
let eventsRemoteInitialized = false;
let eventsSeedAttemptInFlight = false;

function normalizeRealtimeEvent(row){
  if(!row || row.kind === 'meta') return null;
  const id = String(row.id || row._firebaseId || '').trim();
  if(!id) return null;
  const dateISO = String(row.dateISO || '');
  const startTime = String(row.startTime || '');
  const endTime = String(row.endTime || '');
  return {
    id,
    title:String(row.title || 'Etkinlik'),
    dateISO,
    date:String(row.date || (dateISO ? formatAdminDate(dateISO) : '')),
    startTime,
    endTime,
    time:String(row.time || (startTime ? (endTime ? `${startTime} - ${endTime}` : startTime) : '')),
    location:String(row.location || ''),
    capacity:String(row.capacity || ''),
    desc:String(row.desc || ''),
    images:Array.isArray(row.images) ? row.images.slice(0,3).map(String) : [],
    updatedAt:Number(row.updatedAt || 0)
  };
}

function rerenderEventsAfterRealtimeUpdate(){
  const active = document.querySelector('.screen.active');
  if(active && active.id === 'events') renderEventsList();
  if(active && active.id === 'adminEvents') renderAdminEvents();
  if(active && active.id === 'dayAgenda'){
    const chip = document.querySelector('#dayHeader .day-chip');
    const m = chip && String(chip.textContent || '').match(/(\d{1,2})/);
    if(m) openAgendaDay(m[1]);
  }
  if(active && ['eventDetail','eventChat'].includes(active.id) && currentEvent && !events[currentEvent]){
    toast('Bu etkinlik artık yayında değil');
    show('events');
  }else if(active && active.id === 'eventDetail' && currentEvent && events[currentEvent]){
    renderEventDetail();
  }
}

function applyRealtimeEvents(rows, initialized){
  eventsRemoteInitialized = initialized === true;
  const mapped = {};
  (rows || []).forEach(row=>{
    const e = normalizeRealtimeEvent(row);
    if(e) mapped[e.id] = e;
  });

  // Firestore daha önce başlatıldıysa boş liste de gerçektir. İlk kurulumdan önce ise
  // mevcut yerel etkinlikleri admin ilk girişte merkezi koleksiyona taşıyabilsin.
  if(eventsRemoteInitialized || Object.keys(mapped).length){
    events = mapped;
    try{ persistEvents(); }catch(_e){}
    rerenderEventsAfterRealtimeUpdate();
    refreshNotificationChatSubscriptions();
  }
}

async function ensureEventsSync(){
  if(eventsSyncStarted) return;
  eventsSyncStarted = true;
  try{
    const api = await waitForPairaFirebaseApi();
    eventsRealtimeUnsub = api.subscribeEvents((rows, initialized)=>{
      applyRealtimeEvents(rows, initialized);
    }, err=>{
      console.warn('PAIRA realtime events sync', err);
      eventsSyncStarted = false;
    });
  }catch(err){
    console.warn('PAIRA realtime events start', err);
    eventsSyncStarted = false;
  }
}

async function ensureEventsSeededForAdmin(){
  if(eventsSeedAttemptInFlight || eventsRemoteInitialized) return;
  eventsSeedAttemptInFlight = true;
  try{
    const api = await waitForPairaFirebaseApi();
    const seeded = await api.seedEventsIfNeeded(Object.values(events || {}));
    if(seeded) eventsRemoteInitialized = true;
  }catch(err){
    // Admin oturumu yoksa sessizce bekle. Admin paneli açıldığında tekrar denenir.
    if(!String(err?.code || '').includes('permission-denied')) console.warn('PAIRA event seed', err);
  }finally{
    eventsSeedAttemptInFlight = false;
  }
}

function persistEvents(){
  try{
    localStorage.setItem('paira_events_v10', JSON.stringify(events));
  }catch(err){
    toast('Etkinlik görselleri çok büyük. Daha az veya daha küçük fotoğraf deneyin.');
    throw err;
  }
}

function formatAdminDate(dateISO){
  if(!dateISO) return '';
  const months = ['Ocak','Şubat','Mart','Nisan','Mayıs','Haziran','Temmuz','Ağustos','Eylül','Ekim','Kasım','Aralık'];
  const p = dateISO.split('-');
  if(p.length!==3) return dateISO;
  return `${parseInt(p[2],10)} ${months[parseInt(p[1],10)-1]} ${p[0]}`;
}

function handleAdminEventPhotos(event){
  const files = Array.from(event.target.files || []);
  const remaining = Math.max(0, 3 - adminEventPhotos.length);
  if(remaining <= 0){
    toast('En fazla 3 fotoğraf eklenebilir');
    event.target.value = '';
    return;
  }

  const chosen = files.slice(0, remaining);
  let done = 0;
  chosen.forEach(file=>{
    const reader = new FileReader();
    reader.onload = ev=>{
      compressProfileImage(ev.target.result, compressed=>{
        adminEventPhotos.push(compressed);
        done++;
        if(done === chosen.length){
          renderAdminEventPhotoGrid();
          event.target.value = '';
          if(files.length > remaining) toast('3 fotoğraf sınırı uygulandı');
        }
      });
    };
    reader.readAsDataURL(file);
  });
}

async function saveAdminEvent(){
  const title = getField('ae_title');
  const dateISO = getField('ae_date');
  const startTime = getField('ae_start');
  const endTime = getField('ae_end');
  const location = getField('ae_location');
  const capacity = getField('ae_capacity');
  const desc = getField('ae_desc');

  if(!title || !dateISO || !startTime || !location){
    toast('Ad, tarih, başlangıç saati ve mekan gerekli');
    return;
  }

  const id = adminEditingEventId || ('event_' + Date.now());
  const wasEdit = !!adminEditingEventId;
  const previous = events[id] || {};

  try{
    const api = await waitForPairaFirebaseApi();

    // İlk merkezi kurulumdaki eski etkinlik aktarımı yeni etkinliğin kaydını bloke etmesin.
    // Aktarım başarısız olsa bile yeni etkinliği ayrıca kaydetmeyi deniyoruz.
    try{
      await api.seedEventsIfNeeded(Object.values(events || {}));
    }catch(seedErr){
      console.warn('PAIRA legacy event seed skipped', seedErr);
    }

    // Yeni seçilen yerel görselleri Cloudinary URL'lerine dönüştür. Mevcut URL'ler aynen kalır.
    const syncedPhotos = await api.uploadEventPhotos(id, adminEventPhotos);
    const nextEvent = {
      ...previous,
      id,
      title,
      dateISO,
      date: formatAdminDate(dateISO),
      startTime,
      endTime,
      time: endTime ? `${startTime} - ${endTime}` : startTime,
      location,
      capacity: capacity || 'Kapasite belirtilmedi',
      desc,
      images:[...syncedPhotos],
      updatedAt:Date.now()
    };

    await api.saveEvent(id, nextEvent);
    events[id] = nextEvent;
    eventsRemoteInitialized = true;
    if(!state.eventMessages[id]) state.eventMessages[id] = [];
    persistEvents();
    persist();

    closeModal();
    renderEventsList();
    renderAdminEvents();
    toast(wasEdit ? 'Etkinlik tüm üyelerde güncellendi ✓' : 'Etkinlik tüm üyelere yayınlandı ✓');

    await publishSystemNotification({
      type: wasEdit ? 'event_update' : 'event_new',
      title: wasEdit ? `${title} güncellendi` : `Yeni etkinlik: ${title}`,
      body: `${formatAdminDate(dateISO)} • ${startTime}${location ? ' • '+location : ''}`,
      eventId:id
    });
    refreshNotificationChatSubscriptions();
    adminEditingEventId = null;
  }catch(err){
    console.warn('PAIRA realtime event save', err);
    const detail = String(err?.code || err?.message || 'Bilinmeyen Firebase hatası');
    modal(`
      <h2 style="color:#e7bd72;margin-top:0">Etkinlik kaydedilemedi</h2>
      <p class="muted" style="line-height:1.55">Firebase hatası: <b>${escapeHtml(detail)}</b></p>
      <button class="gold" onclick="closeModal()">Tamam</button>
    `);
  }
}

async function deleteAdminEvent(id){
  const e = events[id];
  if(!e) return;
  if(!confirm(`"${e.title}" etkinliğini yayından kaldırmak istiyor musun?`)) return;

  try{
    const api = await waitForPairaFirebaseApi();
    await api.seedEventsIfNeeded(Object.values(events || {}));
    await api.deleteEvent(id);

    delete events[id];
    if(state.joinedEvents) delete state.joinedEvents[id];
    if(state.eventMessages) delete state.eventMessages[id];
    persistEvents();
    persist();

    renderEventsList();
    renderAdminEvents();
    toast('Etkinlik tüm üyelerden kaldırıldı ✓');
    await publishSystemNotification({
      type:'event_removed',
      title:`Etkinlik kaldırıldı: ${e.title}`,
      body:`${e.date || ''}${e.location ? ' • '+e.location : ''}`,
      eventId:id
    });
    refreshNotificationChatSubscriptions();
  }catch(err){
    console.warn('PAIRA realtime event delete', err);
    toast('Etkinlik yayından kaldırılamadı');
  }
}

function escapeAttr(s){
  return escapeHtml(String(s || '')).replace(/`/g,'&#96;');
}



function setDevicePartnerIdentity(rec, partnerIndex){
  const partnerName = (rec.partnerNames || [])[partnerIndex] || `${partnerIndex+1}. Partner`;
  localStorage.setItem('paira_partner_identity_v14', JSON.stringify({
    memberId: rec.id || rec.inviteCode || rec._firebaseId,
    inviteCode: rec.inviteCode || rec._firebaseId || '',
    coupleName: rec.name,
    city: rec.city || '',
    partnerIndex,
    partnerName
  }));

  // Davet hesabı açıldığında telefondaki örnek profil adı yerine gerçek çift adını kullan.
  state.profile = state.profile || {};
  if(rec.name) state.profile.name = rec.name;
  if(rec.city && (!state.profile.city || state.profile.city === 'Bodrum')) state.profile.city = rec.city;
}

function getDevicePartnerIdentity(){
  try{return JSON.parse(localStorage.getItem('paira_partner_identity_v14') || 'null');}
  catch(err){return null;}
}

function currentChatIdentity(){
  const p = getDevicePartnerIdentity();
  if(p && p.partnerName) return {partnerName:p.partnerName,coupleName:p.coupleName || 'Çift'};
  const coupleName = (state.profile && state.profile.name) || 'Çift';
  const parts = String(coupleName).split('&').map(x=>x.trim());
  return {partnerName:parts[0] || 'Partner',coupleName};
}

function messageSenderLabel(m){
  if(m.partnerName && m.coupleName) return `${m.partnerName} • ${m.coupleName}`;
  if(m.name) return m.name;
  return 'Üye';
}

function isMyMessage(m){
  const me=currentChatIdentity();
  return m.partnerName===me.partnerName && m.coupleName===me.coupleName;
}

function getDeviceId(){
  let id = localStorage.getItem('paira_device_id_v1');
  if(!id){
    id = 'dev_' + Math.random().toString(36).slice(2) + Date.now().toString(36);
    localStorage.setItem('paira_device_id_v1', id);
  }
  return id;
}

function generateInviteCode(){
  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  let code = '';
  do{
    code = 'PAIRA-';
    for(let i=0;i<6;i++) code += chars[Math.floor(Math.random()*chars.length)];
  }while((state.adminMembers || []).some(m => m.inviteCode === code));
  return code;
}

async function createMemberInvite(){
  const partner1 = getField('mi_partner1');
  const partner2 = getField('mi_partner2');
  const city = getField('mi_city');
  const contact1 = getField('mi_contact1');
  const contact2 = getField('mi_contact2');

  if(!partner1 || !partner2 || !contact1 || !contact2){
    toast('İki partnerin adı ve telefon numarası gerekli');
    return;
  }

  const name = `${partner1} & ${partner2}`;
  const code = generateInviteCode();
  const rec = {
    id:'member_' + Date.now(),
    name,
    partnerNames:[partner1, partner2],
    city,
    contact: contact1,
    contact2: contact2,
    partnerPhones:[contact1, contact2],
    status:'pending',
    inviteCode:code,
    usedDevices:[],
    partnerDevices:['',''],
    maxUses:2,
    createdAt:new Date().toISOString()
  };

  try{
    if(!window.PAIRA_FIREBASE_API) throw new Error('Firebase henüz hazır değil.');
    await window.PAIRA_FIREBASE_API.saveInvitation(code, rec);
    state.adminMembers = state.adminMembers || [];
    state.adminMembers.unshift(rec);
    closeModal();
    renderAdminMembers();
    showInviteCreated(rec.id);
    toast('Davet Firebase’e kaydedildi ✓');
  }catch(err){
    console.warn('Create invitation', err);
    toast('Davet kaydedilemedi');
  }
}

async function copyInviteCode(code){
  try{
    await navigator.clipboard.writeText(code);
    toast('Davet kodu kopyalandı');
  }catch(err){
    const ta = document.createElement('textarea');
    ta.value = code;
    document.body.appendChild(ta);
    ta.select();
    document.execCommand('copy');
    ta.remove();
    toast('Davet kodu kopyalandı');
  }
}


function normalizePhone(phone){
  let digits = String(phone || '').replace(/\D/g,'');
  if(digits.startsWith('0')) digits = '90' + digits.slice(1);
  if(digits.length === 10 && digits.startsWith('5')) digits = '90' + digits;
  return digits;
}

function partnerPhoneFor(m, partner){
  const arr = m.partnerPhones || [m.contact || '', m.contact2 || ''];
  return arr[(partner || 1) - 1] || '';
}

function isPublicWeb(){
  return location.protocol === 'https:' || location.protocol === 'http:';
}

function inviteLinkFor(code, partner){
  if(!isPublicWeb()) return '';
  const u = new URL(window.location.href);
  u.search = '';
  u.hash = '';
  u.searchParams.set('invite', code);
  if(partner) u.searchParams.set('partner', String(partner));
  return u.toString();
}

function buildInviteMessage(m, partner){
  const link = inviteLinkFor(m.inviteCode, partner);
  const partnerName = (m.partnerNames || [])[partner-1] || `${partner}. partner`;
  const linkText = link || '[Web sitesi yayına alındığında güvenli davet bağlantısı burada oluşacak]';
  return `PAIRA Social Club'a davet edildiniz.

${partnerName}, bu davet sizin telefonunuz içindir.
Davet kodunuz: ${m.inviteCode}

Lütfen kendi telefonunuzdan onay verin:
${linkText}

İki partner de onay verdiğinde ortak çift hesabı açılır. Chat mesajlarında hangi partnerin yazdığı ayrıca görünür.`;
}

function sendInviteSMS(id, partner){
  if(!isPublicWeb()){
    toast('Önce PAIRA web sitesini yayına almalıyız');
    return;
  }
  const m = (state.adminMembers || []).find(x=>x.id===id);
  if(!m) return;
  const phone = normalizePhone(partnerPhoneFor(m, partner));
  if(phone.length < 10){
    toast('Geçerli bir telefon numarası gir');
    return;
  }
  const body = buildInviteMessage(m, partner);
  const sep = /iPhone|iPad|iPod/i.test(navigator.userAgent) ? '&' : '?';
  window.location.href = `sms:+${phone}${sep}body=${encodeURIComponent(body)}`;
}

function sendInviteWhatsApp(id, partner){
  if(!isPublicWeb()){
    toast('Önce PAIRA web sitesini yayına almalıyız');
    return;
  }
  const m = (state.adminMembers || []).find(x=>x.id===id);
  if(!m) return;
  const phone = normalizePhone(partnerPhoneFor(m, partner));
  if(phone.length < 10){
    toast('Geçerli bir telefon numarası gir');
    return;
  }
  const body = buildInviteMessage(m, partner);
  window.open(`https://wa.me/${phone}?text=${encodeURIComponent(body)}`, '_blank');
}

  if(!isPublicWeb()){
    toast('Önce PAIRA web sitesini yayına almalıyız');

    document.body.appendChild(ta);
    ta.select();

    ta.remove();
    toast('Davet bağlantısı kopyalandı');
  }
}

async function approveMember(id){

async function removeMember(id){

    renderAdminMembers();
    toast('Üyelikten çıkarıldı');

}

async function deleteInvite(id){
  const idx = (state.adminMembers || []).findIndex(x=>x.id===id);
  if(idx < 0) return;
  const m = state.adminMembers[idx];
  if(!confirm('Bu kayıt silinsin mi?')) return;
  try{
    await window.PAIRA_FIREBASE_API.removeInvitation(m.inviteCode || m._firebaseId);
    state.adminMembers.splice(idx,1);
    renderAdminMembers();
    toast('Kayıt Firebase’den silindi');
  }catch(err){ toast('Firebase silme işlemi başarısız'); }
}

  const statusText = m.status === 'active' ? 'Aktif' : (m.status === 'removed' ? 'Çıkarıldı' : 'Bekliyor');

  if(m.status === 'pending'){

      <button class="editbtn" onclick="showInviteCreated('${m.id}')">Davet Bilgisi</button>

  }

  return `<div class="member-admin-row">
    <div class="row" style="justify-content:space-between;align-items:flex-start">
      <div>
        <div class="member-title">${escapeHtml(m.name)}</div>
        <div class="member-meta">

          ${escapeHtml((m.partnerNames || [])[0] || '1. Partner')}: ${escapeHtml(phones[0] || '-')} • ${p1Done ? 'Onay verdi ✓' : 'Bekliyor'}<br>

function renderAdminMembers(){
  const all = state.adminMembers || [];
  const pending = all.filter(m=>m.status==='pending');
  const active = all.filter(m=>m.status==='active');
  const removed = all.filter(m=>m.status==='removed');

  const count = document.getElementById('adminMemberCount');
  if(count) count.textContent = active.length;

  const p = document.getElementById('adminPendingMembers');
  const a = document.getElementById('adminActiveMembers');
  const r = document.getElementById('adminRemovedMembers');

  if(p) p.innerHTML = pending.length ? pending.map(memberAdminCard).join('') : '<div class="card muted">Bekleyen davet yok.</div>';
  if(a) a.innerHTML = active.length ? active.map(memberAdminCard).join('') : '<div class="card muted">Aktif üye yok.</div>';
  if(r) r.innerHTML = removed.length ? removed.map(memberAdminCard).join('') : '<div class="card muted">Çıkarılan üye yok.</div>';
}

let eventAttendanceUnsub = null;
let eventAttendanceKey = '';

function stopEventAttendanceSync(){
  if(typeof eventAttendanceUnsub === 'function'){
    try{ eventAttendanceUnsub(); }catch(_){ }
  }
  eventAttendanceUnsub = null;
  eventAttendanceKey = '';
}

function applyEventAttendance(eventId, remote){
  state.joinedEvents = state.joinedEvents || {};
  if(remote && remote.joined === true) state.joinedEvents[eventId] = true;
  else delete state.joinedEvents[eventId];
  persist();
  if(currentEvent === eventId){
    const active = document.querySelector('.screen.active');
    if(active && active.id === 'eventDetail') renderEventDetail();
  }
}

async function ensureEventAttendanceSync(eventId){
  const accountId = sharedProfileAccountId();
  if(!eventId || !accountId) return;
  const key = `${eventId}__${accountId}`;
  if(eventAttendanceKey === key && eventAttendanceUnsub) return;

  stopEventAttendanceSync();
  eventAttendanceKey = key;

  try{
    const api = await waitForPairaFirebaseApi();
    const existing = await api.getEventAttendance(eventId, accountId);
    applyEventAttendance(eventId, existing);
    eventAttendanceUnsub = api.subscribeEventAttendance(eventId, accountId, (row)=>{
      applyEventAttendance(eventId, row);
    }, (err)=>{
      console.warn('PAIRA event attendance sync', err);
    });
  }catch(err){
    console.warn('PAIRA event attendance start', err);
  }
}

function applyClubMembership(key, remote){
  const me = currentChatIdentity();
  state.clubMembers[key] = state.clubMembers[key] || [];
  state.clubMembers[key] = state.clubMembers[key].filter(n => n !== currentUser && n !== me.coupleName);

  persist();
  renderClubs();
  if(currentClub === key){
    const active = document.querySelector('.screen.active');
    if(active && active.id === 'clubDetail') renderClubDetail();
  }
}

async function ensureClubMembershipSync(key){
  const accountId = sharedProfileAccountId();
  if(!key || !accountId || clubMembershipUnsubs.has(key)) return;
  try{
    const api = await waitForPairaFirebaseApi();
    const eventId = clubMembershipEventId(key);
    const existing = await api.getEventAttendance(eventId, accountId);
    applyClubMembership(key, existing);
    const unsub = api.subscribeEventAttendance(eventId, accountId, row=>{
      applyClubMembership(key, row);
    }, err=>{
      console.warn('PAIRA club membership sync', key, err);
      clubMembershipUnsubs.delete(key);
    });
    clubMembershipUnsubs.set(key, unsub);
  }catch(err){
    console.warn('PAIRA club membership start', key, err);
  }
}

function ensureAllClubMembershipSync(){

}

  function fsEncodeValue(v){
    if(v === null || typeof v === "undefined") return {nullValue:null};
    if(typeof v === "string") return {stringValue:v};
    if(typeof v === "boolean") return {booleanValue:v};
    if(typeof v === "number"){
      return Number.isInteger(v) ? {integerValue:String(v)} : {doubleValue:v};
    }
    if(Array.isArray(v)) return {arrayValue:{values:v.map(fsEncodeValue)}};
    if(typeof v === "object"){
      const fields = {};
      Object.entries(v).forEach(([k,val]) => {
        if(typeof val !== "undefined") fields[k] = fsEncodeValue(val);
      });
      return {mapValue:{fields}};
    }
    return {stringValue:String(v)};
  }

  function fsDecodeValue(v){
    if(!v || typeof v !== "object") return null;
    if("stringValue" in v) return v.stringValue;
    if("booleanValue" in v) return v.booleanValue;
    if("integerValue" in v) return Number(v.integerValue);
    if("doubleValue" in v) return Number(v.doubleValue);
    if("timestampValue" in v) return v.timestampValue;
    if("nullValue" in v) return null;
    if("arrayValue" in v) return (v.arrayValue.values || []).map(fsDecodeValue);
    if("mapValue" in v){
      const o = {};
      Object.entries(v.mapValue.fields || {}).forEach(([k,val]) => o[k] = fsDecodeValue(val));
      return o;
    }
    return null;
  }

    const ref = doc(db, "invitations", id);
    return await runTransaction(db, async (tx) => {
      const snap = await tx.get(ref);
      if(!snap.exists()) throw new Error("INVITE_NOT_FOUND");

      const data = snap.data() || {};
      if(data.status === "removed") throw new Error("INVITE_REMOVED");

      const devices = Array.isArray(data.partnerDevices) ? [...data.partnerDevices] : ["", ""];
      while(devices.length < 2) devices.push("");
      const otherIndex = partnerIndex === 0 ? 1 : 0;

      if(devices[otherIndex] === deviceId && devices[partnerIndex] !== deviceId){
        throw new Error("DEVICE_OTHER_PARTNER");
      }
      if(devices[partnerIndex] && devices[partnerIndex] !== deviceId){
        throw new Error("PARTNER_ALREADY_OTHER_DEVICE");
      }

      // Bu telefon zaten doğru partner olarak kayıtlıysa tekrar write yapma.
      if(devices[partnerIndex] === deviceId){
        return {...data, partnerDevices:devices, _firebaseId:snap.id};
      }

      devices[partnerIndex] = deviceId;
      const usedDevices = devices.filter(Boolean);
      const status = usedDevices.length >= 2 ? "active" : "pending";

      tx.update(ref, {partnerDevices:devices, usedDevices, status});

            if (data.localId !== "lVOVhXoi5re2NrEzYW1mjCeKDw93") {
              throw new Error("Bu hesap yönetici olarak yetkilendirilmemiş.");
            }
            pairaRestIdToken = data.idToken || "";
            verifiedAdmin = true;
            window.closeModal();
            openAdminTarget(targetScreen);
            if (window.toast) window.toast("Yönetici girişi başarılı");
          } finally {
            clearTimeout(timer);
          }
        } else {
          const timeout = new Promise((_, reject) =>
            setTimeout(() => reject(new Error("Giriş zaman aşımına uğradı.")), 12000)
          );
          await Promise.race([signInWithEmailAndPassword(auth, email, password), timeout]);
          verifiedAdmin = await Promise.race([
            isCurrentUserAdmin(),
            new Promise((_, reject) =>
              setTimeout(() => reject(new Error("Yönetici kontrolü zaman aşımına uğradı.")), 12000)
            )
          ]);

          if (!verifiedAdmin) {
            await signOut(auth);
            throw new Error("Bu hesap yönetici olarak yetkilendirilmemiş.");
          }

          window.closeModal();
          openAdminTarget(targetScreen);
          if (window.toast) window.toast("Yönetici girişi başarılı");
        }
      } catch (err) {
        const msg = err?.message || "";
        errEl.textContent =
          /INVALID_LOGIN_CREDENTIALS|INVALID_PASSWORD|EMAIL_NOT_FOUND|invalid-credential/i.test(msg)
            ? "E-posta veya şifre hatalı."
            : /aborted|zaman aşımı|timeout/i.test(msg)
              ? "Bağlantı zaman aşımına uğradı. İnterneti kontrol edip tekrar dene."
              : (msg || "Giriş yapılamadı.");
        errEl.style.display = "block";
      } finally {
        btn.disabled = false;
        btn.textContent = "Giriş Yap";
      }
    });

    setTimeout(() => {
      if (!emailEl.value) emailEl.focus();
      else passEl.focus();
    }, 50);
  }

  async function publishCurrentWeb(){
    const payload = {
      html: PAIRA_WEB_SOURCE,
      version: "v14-firebase-web",
      updatedAt: new Date().toISOString()
    };

    const bytes = new Blob([payload.html]).size;
    if(bytes > 900000){
      throw new Error("Web dosyası Firebase belge sınırına fazla yaklaştı.");
    }

    if(isLocalContent()){
      await restFirestore("webdeploy/current", {
        method:"PATCH",
        body:JSON.stringify({fields:fsEncodeFields(payload)})
      });
    }else{
      await setDoc(doc(db, "webdeploy", "current"), payload);
    }

    modal(`
      <h2 style="color:#e7bd72;margin-top:0">Web Kaynağı Hazır ✓</h2>
      <p class="muted" style="line-height:1.55">
        PAIRA V14 Firebase sürümü web yayını için Firebase’e yüklendi.
      </p>
      <button class="gold" onclick="closeModal()">Tamam</button>
    `);
  }

  async function openAdminTarget(targetScreen) {
    if (targetScreen === "webPublish") {
      try{
        await publishCurrentWeb();
        if(window.toast) window.toast("Web kaynağı Firebase’e yüklendi ✓");
      }catch(err){
        console.warn("PAIRA web publish", err);
        modal(`
          <h2 style="color:#e7bd72;margin-top:0">Yayınlama Hatası</h2>
          <p class="muted" style="line-height:1.55">${String(err?.message || "Web kaynağı yüklenemedi.")}</p>
          <button class="gold" onclick="closeModal()">Tamam</button>
        `);
      }
      return;
    }
    if (targetScreen === "adminMembers") {
      if (window.show) window.show(targetScreen);
      try{
        if(window.pairaLoadInvitations) await window.pairaLoadInvitations();
      }catch(err){
        console.warn("PAIRA invitation load", err);
        if(window.toast) window.toast("Davetler Firebase'den alınamadı");
      }
      return;
    }
    if (targetScreen === "adminEvents" && window.renderAdminEvents) window.renderAdminEvents();
    if (targetScreen === "adminInbox" && window.renderAdminInbox) window.renderAdminInbox();
    if (window.show) window.show(targetScreen);
  }

  window.pairaOpenAdminLogin = async function() {
    verifiedAdmin = false;
    pairaRestIdToken = "";
    try {
      if (auth.currentUser) await signOut(auth);
    } catch (_) {}
    openAdminLogin('adminPanel');
  };

  window.pairaRequireAdmin = async function(targetScreen) {
    if (verifiedAdmin && auth.currentUser) {
      openAdminTarget(targetScreen);
      return;
    }

    verifiedAdmin = await isCurrentUserAdmin();
    if (verifiedAdmin) {
      openAdminTarget(targetScreen);
      return;
    }

    openAdminLogin(targetScreen);
  };

  window.pairaAdminLogout = async function() {
    verifiedAdmin = false;
    pairaRestIdToken = "";
    await signOut(auth);
    if (window.toast) window.toast("Yönetici oturumu kapatıldı");
  };

  onAuthStateChanged(auth, async (user) => {
    verifiedAdmin = user ? await isCurrentUserAdmin() : false;
  });

  window.PAIRA_FIREBASE = { app, auth, db, storage, config: firebaseConfig };
  window.dispatchEvent(new CustomEvent("paira-firebase-ready"));
